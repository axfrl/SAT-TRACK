   483                                                   y1 = (cy - h / 2) * img_height
   484                                                   x2 = (cx + w / 2) * img_width
   485                                                   y2 = (cy + h / 2) * img_height
   486                                                   pred_bbox = torch.stack([x1, y1, x2, y2], dim=-1).cpu().numpy()
   487                                           
   488                                                   # Filter detections based on score and size thresholds
   489                                                   NPEOPLE = len(pred_confs)
   490                                                   masked_image_list = []
   491                                                   center_list = []
   492                                                   scale_list = []
   493                                                   rles_list = []
   494                                                   selected_ids = []
   495                                                   pred_classes = np.zeros(NPEOPLE, dtype=np.int64)  # Class 0 (person)
   496                                                   ground_truth_track_id = [1] * NPEOPLE  # Default track IDs
   497                                                   ground_truth_annotations = [[]] * NPEOPLE  # Default annotations
   498                                           
   499                                                   for p_ in range(NPEOPLE):
   500                                                       if pred_confs[p_] < self.cfg.phalp.low_th_c:
   501                                                           continue
   502                                                       w = pred_bbox[p_][2] - pred_bbox[p_][0]
   503                                                       h = pred_bbox[p_][3] - pred_bbox[p_][1]
   504                                                       if w < self.cfg.phalp.small_w or h < self.cfg.phalp.small_h:
   505                                                           continue
   506                                                       
   507                                                       # Generate synthetic mask
   508                                                       mask = self._generate_synthetic_masks(pred_bbox[p_][None], img_height, img_width)[0]
   509                                                       rle = mask_utils.encode(np.asfortranarray(mask.astype(np.uint8)))
   510                                                       rle['counts'] = rle['counts'].decode('utf-8')
   511                                           
   512                                                       # Compute center, scale, and cropped image for HMAR
   513                                                       center = np.array([(pred_bbox[p_][0] + pred_bbox[p_][2]) / 2, (pred_bbox[p_][1] + pred_bbox[p_][3]) / 2])
   514                                                       scale = np.array([w, h]) / self.cfg.render.res
   515                                                       bbox = pred_bbox[p_]
   516                                                       bbox_pad = np.array([bbox[0] - w * 0.125, bbox[1] - h * 0.125, bbox[2] + w * 0.125, bbox[3] + h * 0.125])
   517                                                       masked_image, center_pad, scale_pad, _, _, _ = self.get_croped_image(image, bbox, bbox_pad, mask)
   518                                                       
   519                                                       masked_image_list.append(masked_image)
   520                                                       center_list.append(center_pad)
   521                                                       scale_list.append(scale_pad)
   522                                                       rles_list.append(rle)
   523                                                       selected_ids.append(p_)
   524                                                   
   525                                                   if len(selected_ids) == 0:
   526                                                       return []
   527                                           
   528                                                   BS = len(selected_ids)
   529                                                   
   530                                                   # Run HMAR for appearance and UV maps
   531                                                   if BS > 0:
   532                                                       masked_image_list = torch.stack(masked_image_list, dim=0)
   533                                                       with torch.no_grad():
   534                                                           extra_args = {}
   535                                                           hmar_out = self.HMAR(masked_image_list.cuda(), **extra_args)
   536                                                           uv_vector = hmar_out['uv_vector']  # [BS, 256, 256, 3]
   537                                                           appe_embedding = self.HMAR.autoencoder_hmar(uv_vector, en=True)  # [BS, embedding_dim]
   538                                                           appe_embedding  = appe_embedding.view(appe_embedding.shape[0], -1)
   539                                           
   540                                                   else:
   541                                                       uv_vector = np.zeros((BS, 256, 256, 3))
   542                                                       appe_embedding = torch.zeros(BS, 512)  # Dummy embedding
   543                                           
   544                                                   # Prepare SMPL parameters and joints from SAT-HMR
   545                                                   pred_cam_tensor = pred_transl[selected_ids]  # (BS, 3) – torch.Tensor
   546                                                   pred_smpl_params = []
   547                                                   for idx in selected_ids:
   548                                                       aa        = pred_poses[idx]            # (72,)
   549                                                       global_aa = aa[:3].unsqueeze(0)        # (1,3)
   550                                                       body_aa   = aa[3:].view(-1,3)          # (23,3)
   551                                           
   552                                                       # axis-angle → rotmat
   553                                                       global_rot = batch_rodrigues(global_aa)[0].cpu().numpy()   # (3,3)
   554                                                       body_rot   = batch_rodrigues(body_aa).cpu().numpy()       # (23,3,3)
   555                                           
   556                                                       pred_smpl_params.append({
   557                                                           'global_orient': global_rot[None, ...],  # (1,3,3)
   558                                                           'body_pose'    : body_rot,               # (23,3,3)
   559                                                           'betas'        : pred_betas[idx].cpu().numpy()  # (10,)
   560                                                       })
   561                                                   
   562                                                   pred_cam_np = self.transl3d_to_weak_cam(pred_cam_tensor.cpu().numpy())  # (BS,3) array
   563                                           
   564                                                   pred_joints_3d = pred_j3ds[selected_ids].cpu().numpy()  # [BS, num_joints, 3]
   565                                                   pred_joints_2d = pred_j2ds[selected_ids].cpu().numpy()  # [BS, num_joints, 2]
   566                                                   pred_cam = pred_transl[selected_ids].cpu().numpy()  # [BS, 3]
   567                                                   pred_cam_weak = []
   568                                                   # Compute pose embedding
   569                                                   if self.cfg.phalp.pose_distance == "joints":
   570                                                       pose_embedding = torch.from_numpy(pred_joints_3d).view(BS, -1)
   571                                                   elif self.cfg.phalp.pose_distance == "smpl":
   572                                                       pose_embedding_list = []
   573                                                       for i in range(BS):
   574                                                           pred_cam_weak.append(pred_cam_np[i])
   575                                                           emb_np = smpl_to_pose_camera_vector(
   576                                                               pred_smpl_params[i],      # dict avec rotmats + betas
   577                                                               pred_cam_np[i]            # [scale, tx, ty]
   578                                                           )
   579                                                           pose_embedding_list.append(torch.from_numpy(emb_np[0]))
   580                                                       pose_embedding = torch.stack(pose_embedding_list, dim=0)
   581                                                   else:
   582                                                       raise ValueError("Unknown pose distance")
   583                                                   
   584                                                   # Compute location embedding
   585                                                   pred_joints_2d_ = pred_joints_2d.reshape(BS, -1) / self.cfg.render.res
   586                                                   pred_cam_ = pred_cam.reshape(BS, -1)
   587                                                   loca_embedding = torch.cat((torch.from_numpy(pred_joints_2d_), torch.from_numpy(pred_cam_),
   588                                                                           torch.from_numpy(pred_cam_), torch.from_numpy(pred_cam_)), dim=1)
   589                                           
   590                                                   # Compute full embedding (for legacy)
   591                                                   full_embedding = torch.cat((appe_embedding.cpu(), pose_embedding, loca_embedding), dim=1)
   592                                                   
   593                                                   j2ds_smpl = pred_j2ds[selected_ids].cpu().numpy()  # [BS, 45, 2]
   594                                                   # convertis en 17 keypoints COCO
   595                                                   j2ds_coco = smpl45_to_coco17(j2ds_smpl)            # [BS, 17, 2]
   596                                           
   597                                                   # Create detection data list
   598                                                   detection_data_list = []
   599                                                   for i, p_ in enumerate(selected_ids):
   600                                                       joints_2d = j2ds_coco[i]
   601                                                       keypoints = []
   602                                                       count_out = 0
   603                                                       for j in range(joints_2d.shape[0]):
   604                                                           x, y = joints_2d[j]
   605                                                           if 0 <= x < img_width and 0 <= y < img_height:
   606                                                               vi = 1
   607                                                           else:
   608                                                               vi = 0
   609                                                               count_out += 1
   610                                                           keypoints.extend([x, y, vi])
   611                                                       
   612                                                       detection_data = {
   613                                                           "bbox": np.array([pred_bbox[p_][0], pred_bbox[p_][1],
   614                                                                           pred_bbox[p_][2] - pred_bbox[p_][0], pred_bbox[p_][3] - pred_bbox[p_][1]]),
   615                                                           "mask": rles_list[i],
   616                                                           "conf": pred_confs[p_].cpu().numpy(),
   617                                                           "appe": appe_embedding[i].cpu().numpy(),
   618                                                           "pose": pose_embedding[i].numpy(),
   619                                                           "loca": loca_embedding[i].numpy(),
   620                                                           "uv": uv_vector[i],
   621                                                           "embedding": full_embedding[i].numpy(),
   622                                                           "center": center_list[i] + np.array([left, top]) * ratio,
   623                                                           "scale": scale_list[i] * ratio,
   624                                                           "smpl": pred_smpl_params[i],
   625                                                           "camera": pred_cam_[i],
   626                                                           "camera_bbox": pred_cam_[i],  # Use transl as proxy
   627                                                           "pred_intrinsic": pred_intrinsic,
   628                                                           "pred_cam_xys": pred_cam_xys,
   629                                                           "3d_joints": pred_joints_3d[i],
   630                                                           "2d_joints": pred_joints_2d_[i],
   631                                                           "size": [img_height, img_width],
   632                                                           "img_path": frame_name,
   633                                                           "img_name": frame_name.split('/')[-1] if isinstance(frame_name, str) else None,
   634                                                           "class_name": pred_classes[p_],
   635                                                           "time": t_,
   636                                                           "ground_truth": gt[p_] if gt is not None else ground_truth_track_id[p_],
   637                                                           "annotations": ann[p_] if ann is not None else ground_truth_annotations[p_],
   638                                                           "extra_data": extra_data[p_] if extra_data is not None else None,
   639                                                           "keypoints": keypoints
   640                                                       }
   641                                                       detection_data_list.append(Detection(detection_data))
   642                                           
   643                                                   return detection_data_list
